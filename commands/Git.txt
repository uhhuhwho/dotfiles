alias dotfiles='git --git-dir=$HOME/.dotfiles --work-tree=$HOME'

Check your Git and Github connection:
ssh -T git@github.com

To add a file:
dotfiles add ~/.dotfiles/README.md

To add a folder:
dotfiles add ~/.dotfiles/

To comit files with a comment:
dotfiles commit -m "Added README.md"

To push changes
dotfiles push

##
dotfiles commit -am "Update"
dotfiles push
